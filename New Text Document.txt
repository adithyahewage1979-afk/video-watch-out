<script>
let ownerPassword = "hptq536";
let isOwner = false;
let pendingVideo = null;
let videos = [];
let nextId = 1;

// Demo video records and file previews remain in browser memory.
const $ = id => document.getElementById(id);

function show(id) {
  ["ownerChoice","login","store","dashboard","checkout"].forEach(
    name => $(name).classList.toggle("hidden", name !== id)
  );
}

function home() {
  renderVideos();
  show("store");
}

function loginOwner(event) {
  event.preventDefault();
  if ($("password").value === ownerPassword) {
    isOwner = true;
    $("password").value = "";
    $("loginError").textContent = "";
    renderManage();
    show("dashboard");
  } else {
    $("loginError").textContent = "Incorrect password. Please try again.";
  }
}

function logout() {
  isOwner = false;
  show("store");
}

function addVideo(event) {
  event.preventDefault();
  if (!isOwner) return alert("Owner login required.");

  const videoFile = $("videoFile").files[0];
  const imageFile = $("imageFile").files[0];

  if (!videoFile || !imageFile) {
    alert("Please select a video and a thumbnail.");
    return;
  }

  if (!videoFile.type.startsWith("video/") ||
      !imageFile.type.startsWith("image/")) {
    alert("Choose a valid video and image file.");
    return;
  }

  const reader = new FileReader();
  reader.onload = () => {
    const thumb = reader.result;
    const videoURL = URL.createObjectURL(videoFile);

    videos.push({
      id: nextId++,
      title: $("title").value.trim(),
      description: $("description").value.trim(),
      price: Number($("price").value),
      thumb,
      videoURL,
      file: videoFile
    });

    $("uploadForm").reset();
    renderVideos();
    renderManage();
    alert("Video added to this demo!");
  };
  reader.readAsDataURL(imageFile);
}

function renderVideos() {
  const query = ($("search")?.value || "").toLowerCase();
  const list = videos.filter(v =>
    v.title.toLowerCase().includes(query) ||
    v.description.toLowerCase().includes(query)
  );

  $("videos").innerHTML = list.length ? list.map(v => `
    <article class="card">
      <img class="thumb" src="${v.thumb}" alt="Video thumbnail">
      <div class="card-body">
        <h3>${escapeHTML(v.title)}</h3>
        <p>${escapeHTML(v.description || "Premium video")}</p>
        <div class="price">Rs. ${v.price.toLocaleString("en-LK")}</div>
        <button class="green" onclick="buy(${v.id})">Buy Now</button>
      </div>
    </article>`).join("") :
    `<div class="notice">No videos yet. ${isOwner ? "Upload your first video in the dashboard." : "Please check back soon."}</div>`;
}

function renderManage() {
  if (!isOwner) return;
  $("manageVideos").innerHTML = videos.length ? videos.map(v => `
    <article class="card card-body">
      <img class="thumb" src="${v.thumb}" alt="Thumbnail">
      <h3>${escapeHTML(v.title)}</h3>
      <p>Rs. ${v.price.toLocaleString("en-LK")}</p>
      <button class="secondary" onclick="editVideo(${v.id})">Edit</button>
      <button class="secondary" onclick="deleteVideo(${v.id})">Delete</button>
    </article>`).join("") : "<p>No videos uploaded yet.</p>";
}

function editVideo(id) {
  if (!isOwner) return;
  const v = videos.find(item => item.id === id);
  if (!v) return;
  const title = prompt("Video title:", v.title);
  if (title === null) return;
  const price = prompt("Price in LKR:", v.price);
  if (price === null) return;
  if (!title.trim() || !Number.isFinite(Number(price)) || Number(price) <= 0) {
    alert("Enter a valid title and price.");
    return;
  }
  v.title = title.trim();
  v.price = Number(price);
  renderManage();
  renderVideos();
}

function deleteVideo(id) {
  if (!isOwner || !confirm("Delete this video from the demo?")) return;
  const index = videos.findIndex(v => v.id === id);
  if (index >= 0) {
    URL.revokeObjectURL(videos[index].videoURL);
    videos.splice(index, 1);
  }
  renderManage();
  renderVideos();
}

function buy(id) {
  pendingVideo = videos.find(v => v.id === id);
  if (!pendingVideo) return;
  $("orderDetails").innerHTML = `
    <h3>${escapeHTML(pendingVideo.title)}</h3>
    <p>Total: <strong>Rs. ${pendingVideo.price.toLocaleString("en-LK")}</strong></p>`;
  $("paymentMsg").textContent = "";
  show("checkout");
}

function demoPayment() {
  $("paymentMsg").textContent =
    "Demo only: no payment was made. Connect a real payment gateway to complete purchases.";
}

function changeBrand() {
  if (!isOwner) return;
  const name = $("newBrand").value.trim();
  if (!name) return alert("Enter a website name.");
  $("brand").textContent = "▶ " + name;
  $("newBrand").value = "";
  $("settingsMsg").textContent = "Name changed for this session.";
}

function changePassword() {
  if (!isOwner) return;
  if ($("oldPass").value !== ownerPassword) {
    $("settingsMsg").textContent = "Current password is incorrect.";
    return;
  }
  const next = $("newPass").value;
  if (next.length < 6) {
    $("settingsMsg").textContent = "New password must be at least 6 characters.";
    return;
  }
  ownerPassword = next;
  $("oldPass").value = "";
  $("newPass").value = "";
  $("settingsMsg").textContent = "Password updated for this session only.";
}

function escapeHTML(value) {
  return String(value).replace(/[&<>"']/g, ch => ({
    "&":"&amp;", "<":"&lt;", ">":"&gt;",
    '"':"&quot;", "'":"&#39;"
  })[ch]);
}

$("year").textContent = new Date().getFullYear();
show("ownerChoice");
